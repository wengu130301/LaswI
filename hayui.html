// HayUI · Android 端示例 class MainActivity : HayActivity() { override fun onCreate(savedInstanceState: Bundle?) { super.onCreate(savedInstanceState) setContent { HayTheme { GlassContainer( modifier = Modifier.fillMaxSize(), blur = 20.dp, background = "#1a1a2e" ) { Column { Text("欢迎使用 HayUI", style = HayTheme.typography.h1) Button(onClick = { navigateTo("/settings") }) { Text("打开设置") } } } } } } } // HayUI · Linux 端示例 (Rust + GTK4) use hayui::prelude::*; fn main() -> Result<(), Box<dyn Error>> { HayApp::new() .window(Window::new() .title("HayUI · Linux") .size(800, 600) .glass(GlassConfig::default().blur(20.0)) .child( Column::new() .push(Text::new("欢迎使用 HayUI").style(TextStyle::Heading1)) .push(Button::new("打开设置").on_click(|| { navigate("/settings"); })) ) ) .run() }
